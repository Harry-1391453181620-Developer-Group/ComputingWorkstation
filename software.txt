Instructions to boost full use of harwares in Linux:
sudo apt install -y linux-tools-generic
sudo cpupower frequency-set -b performance
watch -n 1 "cat /proc/cpuinfo | grep -i mhz"
sudo systemctl enable cpupower
sudo nvidia-smi -pm 1
sudo nvidia-smi -lgc 2500,2500
sudo sysctl -w vm.swappiness=0
sudo sysctl -w vm.dirty_background_ratio=5
sudo sysctl -w vm.dirty_ratio=10
sudo nano /etc/default/grub
in the file, do: GRUB_CMDLINE_LINUX_DEFAULT="quiet splash amd_iommu=on iommu=pt processor.max_cstate=1 intel_idle.max_cstate=0 pcie_aspm=performance"
sudo update-grub
sudo reboot
